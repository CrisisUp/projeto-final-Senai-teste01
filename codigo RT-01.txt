! =========================================================
! RT-01 — Roteador da Filial
! LAN: VLAN 1 (PCs) + VLAN 10 (IoT/Wireless) via Router-on-a-Stick
! + ACL IoT + hardening
! =========================================================
enable
configure terminal

! 0. Hardening basico
enable secret Senai@Enable2026
service password-encryption
service timestamps log datetime msec
no ip http server
no ip http secure-server
ip domain-name senai.local
username admin privilege 15 secret Senai@Admin2026

banner motd #
=========================================================
 ATENCAO: Acesso restrito a autorizados do SENAI.
 Equipamento: RT-01 (Filial). Lab academico.
=========================================================
#

crypto key generate rsa general-keys modulus 1024

line console 0
 logging synchronous
 exec-timeout 5 0
 login local
line vty 0 4
 transport input ssh
 login local
line vty 5 15
 transport input ssh
 login local

! 1. Nome do equipamento
hostname RT-01

! 2. Interface fisica conectada ao SW-01 (trunk, sem IP)
interface gigabitEthernet 0/0
 description TRUNK_LOCAL_SW-01
 no shutdown
 exit

! 3. Subinterfaces (Gateways das VLANs da Filial)

! VLAN 1 — Rede local dos PCs (acesso a Matriz/Internet liberado)
interface gigabitEthernet 0/0.1
 description GATEWAY_VLAN01_FILIAL
 encapsulation dot1Q 1
 ip address 192.168.55.1 255.255.255.0
 exit

! VLAN 10 — IoT / Wireless da Filial
! Politica: DNS/servidor; NAO acessa TI da Matriz
interface gigabitEthernet 0/0.10
 description GATEWAY_VLAN10_IOT_FILIAL
 encapsulation dot1Q 10
 ip address 192.168.56.1 255.255.255.0
 ip access-group ACL_FILIAL_IOT_IN in
 exit

! 4. Interface WAN conectada ao RT-00 (borda)
interface gigabitEthernet 0/1
 description LINK_WAN_RT-00
 ip address 10.100.0.2 255.255.255.252
 duplex auto
 speed auto
 no shutdown
 exit

! 5. Rota padrao de saida da Filial apontando para a borda (RT-00)
ip route 0.0.0.0 0.0.0.0 10.100.0.1

! 6. ACL — VLAN 10 IoT da Filial (inbound)
ip access-list extended ACL_FILIAL_IOT_IN
 remark IoT filial: DHCP
 permit udp any eq 68 any eq 67
 remark IoT filial: gateway local
 permit icmp any host 192.168.56.1
 remark IoT filial: DNS/servidor Matriz
 permit udp 192.168.56.0 0.0.0.255 host 192.168.3.10 eq 53
 permit icmp 192.168.56.0 0.0.0.255 host 192.168.3.10
 permit udp 192.168.57.0 0.0.0.255 host 192.168.3.10 eq 53
 permit icmp 192.168.57.0 0.0.0.255 host 192.168.3.10
 remark IoT filial NAO acessa TI da Matriz
 deny   ip 192.168.56.0 0.0.0.255 192.168.8.0 0.0.0.255
 deny   ip 192.168.57.0 0.0.0.255 192.168.8.0 0.0.0.255
 remark Demais destinos
 permit ip any any

! 7. DHCP — VLAN 1 (PCs da Filial)
!    DNS corporativo: servidor Matriz 192.168.3.10
ip dhcp excluded-address 192.168.55.1 192.168.55.20
ip dhcp excluded-address 192.168.56.1 192.168.56.20

ip dhcp pool DHCP_FILIAL_VLAN1
 network 192.168.55.0 255.255.255.0
 default-router 192.168.55.1
 dns-server 192.168.3.10
 domain-name filial.local
 lease 0 8 0

! 8. DHCP — VLAN 10 (dispositivos IoT wireless da Filial)
ip dhcp pool DHCP_FILIAL_VLAN10
 network 192.168.56.0 255.255.255.0
 default-router 192.168.56.1
 dns-server 192.168.3.10
 domain-name iot.local
 lease 1 0 0

! 9. Salvar configuracoes
end
write memory
