! =========================================================
! SW-02 — Switch da Matriz
! VLAN 3  : Servidor Matriz
! VLAN 5  : PC-04 Colaboradores
! VLAN 8  : PC-05 TI
! VLAN 10 : Roteador Wi-Fi IoT
! Uplink  : trunk para RT-02 (Router-on-a-Stick)
! =========================================================
enable
configure terminal

! 1. Nome do comutador
hostname SW-02

! 2. Criacao das VLANs
vlan 3
 name SERVIDORES
 exit

vlan 5
 name COLABORADORES
 exit

vlan 8
 name TI
 exit

vlan 10
 name IOT
 exit

! 3. Uplink para o roteador RT-02 (trunk 802.1Q)
interface gigabitEthernet 0/1
 description UPLINK_TRUNK_RT-02
 switchport mode trunk
 switchport trunk allowed vlan 3,5,8,10
 no shutdown
 exit

! 4. Atribuicao e descricao das portas de acesso

! Servidor Matriz (VLAN 3) — IP estatico 192.168.3.10 (DNS)
interface gigabitEthernet 0/2
 description CONEXAO_SERVIDOR_MATRIZ
 switchport mode access
 switchport access vlan 3
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
 exit

! PC-04 (VLAN 5 - Colaboradores) — recebe DHCP do RT-02
interface fastEthernet 0/4
 description CONEXAO_PC04_COLABORADORES
 switchport mode access
 switchport access vlan 5
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
 exit

! PC-05 (VLAN 8 - TI) — recebe DHCP do RT-02
interface fastEthernet 0/5
 description CONEXAO_PC05_TI
 switchport mode access
 switchport access vlan 8
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
 exit

! Roteador Wi-Fi IoT (VLAN 10) — WAN estatica 192.168.10.2
interface fastEthernet 0/6
 description LINK_ROTEADOR_WIFI_IOT
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
 exit

! 5. Salvar configuracoes
end
write memory
