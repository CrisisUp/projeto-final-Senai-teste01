! =========================================================
! SW-02 — Switch da Matriz
! VLAN 3/5/8/10 + trunk RT-02
! Hardening: port security, DHCP snooping, portas ociosas
! =========================================================
enable
configure terminal

! 0. Hardening basico
enable secret Senai@Enable2026
service password-encryption
service timestamps log datetime msec
no ip http server
no ip http secure-server
ip domain-name senai.local
username admin privilege 15 secret Senai@Admin2026

banner motd #
=========================================================
 ATENCAO: Acesso restrito a autorizados do SENAI.
 Equipamento: SW-02 (Matriz). Lab academico.
=========================================================
#

crypto key generate rsa general-keys modulus 1024

line console 0
 logging synchronous
 exec-timeout 5 0
 login local
line vty 0 4
 transport input ssh
 login local
line vty 5 15
 transport input ssh
 login local

! 1. Nome do comutador
hostname SW-02

! 2. Criacao das VLANs
vlan 3
 name SERVIDORES
 exit

vlan 5
 name COLABORADORES
 exit

vlan 8
 name TI
 exit

vlan 10
 name IOT
 exit

! 3. Uplink para o roteador RT-02 (trunk 802.1Q)
interface gigabitEthernet 0/1
 description UPLINK_TRUNK_RT-02
 switchport mode trunk
 switchport trunk allowed vlan 3,5,8,10
 no shutdown
 exit

! 4. Atribuicao e descricao das portas de acesso

! Servidor Matriz (VLAN 3) — IP estatico 192.168.3.10 (DNS)
interface gigabitEthernet 0/2
 description CONEXAO_SERVIDOR_MATRIZ
 switchport mode access
 switchport access vlan 3
 spanning-tree portfast
 spanning-tree bpduguard enable
 switchport port-security
 switchport port-security maximum 1
 switchport port-security violation shutdown
 switchport port-security mac-address sticky
 no shutdown
 exit

! PC-04 (VLAN 5 - Colaboradores)
interface fastEthernet 0/4
 description CONEXAO_PC04_COLABORADORES
 switchport mode access
 switchport access vlan 5
 spanning-tree portfast
 spanning-tree bpduguard enable
 switchport port-security
 switchport port-security maximum 1
 switchport port-security violation shutdown
 switchport port-security mac-address sticky
 no shutdown
 exit

! PC-05 (VLAN 8 - TI)
interface fastEthernet 0/5
 description CONEXAO_PC05_TI
 switchport mode access
 switchport access vlan 8
 spanning-tree portfast
 spanning-tree bpduguard enable
 switchport port-security
 switchport port-security maximum 1
 switchport port-security violation shutdown
 switchport port-security mac-address sticky
 no shutdown
 exit

! Roteador Wi-Fi IoT (VLAN 10) — WAN estatica 192.168.10.2
interface fastEthernet 0/6
 description LINK_ROTEADOR_WIFI_IOT
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
 spanning-tree bpduguard enable
 switchport port-security
 switchport port-security maximum 1
 switchport port-security violation shutdown
 switchport port-security mac-address sticky
 no shutdown
 exit

! 5. Portas nao utilizadas — shutdown (boas praticas)
interface range fastEthernet 0/7 - 24
 description PORTA_OCIOSA
 shutdown
 exit

! 6. DHCP Snooping
ip dhcp snooping
ip dhcp snooping vlan 3,5,8,10
interface gigabitEthernet 0/1
 ip dhcp snooping trust
 exit

! 7. Salvar configuracoes
end
write memory
