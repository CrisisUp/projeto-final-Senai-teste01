! Definir o nome do equipamento (caso ainda nao esteja definido)
hostname RT-00

! Ativar a interface g0/2 conectada ao RT-02
interface gigabitEthernet 0/2
 description LINK_WAN_RT-02_MATRIZ
 ip address 10.0.0.1 255.255.255.252
 duplex auto
 speed auto
 no shutdown
 exit

! Rotas de retorno para as VLANs da Rede Matriz via RT-02 (10.0.0.2)
ip route 192.168.3.0 255.255.255.0 10.0.0.2
ip route 192.168.5.0 255.255.255.0 10.0.0.2
ip route 192.168.8.0 255.255.255.0 10.0.0.2
ip route 192.168.10.0 255.255.255.0 10.0.0.2

! Guardar configuracoes
end
write memory


! 1. Configuracao da interface WAN conectada a Filial

 duplex auto
 speed auto

end
